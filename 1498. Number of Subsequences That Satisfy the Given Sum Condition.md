

You are given an array of integers `nums` and an integer `target`.

Return _the number of **non-empty** subsequences of_ `nums` _such that the sum of the minimum and maximum element on it is less or equal to_ `target`. Since the answer may be too large, return it **modulo** `109 + 7`.

**Example 1:**

**Input:** nums = [3,5,6,7], target = 9
**Output:** 4
**Explanation:** There are 4 subsequences that satisfy the condition.
[3] -> Min value + max value <= target (3 + 3 <= 9)
[3,5] -> (3 + 5 <= 9)
[3,5,6] -> (3 + 6 <= 9)
[3,6] -> (3 + 6 <= 9)

**Example 2:**

**Input:** nums = [3,3,6,8], target = 10
**Output:** 6
**Explanation:** There are 6 subsequences that satisfy the condition. (nums can have repeated numbers).
[3] , [3] , [3,3], [3,6] , [3,6] , [3,3,6]

**Example 3:**

**Input:** nums = [2,3,3,4,6,7], target = 12
**Output:** 61
**Explanation:** There are 63 non-empty subsequences, two of them do not satisfy the condition ([6,7], [7]).
Number of valid subsequences (63 - 2 = 61).

**Constraints:**

- `1 <= nums.length <= 105`
- `1 <= nums[i] <= 106`
- `1 <= target <= 106`

### recursion

time limit exceeded

```cpp
class Solution {
private:
    int count = 0;

    void helper(int idx, vector<int>& nums, int target, vector<int>& curr) {
        if (idx == nums.size()) {
            if (!curr.empty()) {
                int mn = *min_element(curr.begin(), curr.end());
                int mx = *max_element(curr.begin(), curr.end());

                if (mn + mx <= target)
                    count++;
            }
            return;
        }

        // Choice 1: Take nums[idx]
        curr.push_back(nums[idx]);
        helper(idx + 1, nums, target, curr);
        curr.pop_back();

        // Choice 2: Skip nums[idx]
        helper(idx + 1, nums, target, curr);
    }

public:
    int numSubseq(vector<int>& nums, int target) {
        vector<int> curr;
        helper(0, nums, target, curr);
        return count;
    }
};
```

```cpp
class Solution {
public:
    int numSubseq(vector<int>& nums, int target) {
        const int MOD = 1e9 + 7;

        sort(nums.begin(), nums.end());

        int n = nums.size();

        vector<long long> pow2(n);
        pow2[0] = 1;

        for (int i = 1; i < n; i++) {
            pow2[i] = (pow2[i - 1] * 2) % MOD;
        }

        int left = 0;
        int right = n - 1;

        long long res = 0;

        while (left <= right) {
            if (nums[left] + nums[right] <= target) {
                res = (res + pow2[right - left]) % MOD;
                left++;
            } else {
                right--;
            }
        }

        return res;
    }
};
```


Time:  O(n log n)
Space: O(n)
