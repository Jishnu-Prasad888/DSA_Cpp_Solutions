Given a string containing digits from `2-9` inclusive, return all possible letter combinations that the number could represent. Return the answer in **any order**.

A mapping of digits to letters (just like on the telephone buttons) is given below. Note that 1 does not map to any letters.

![](https://assets.leetcode.com/uploads/2022/03/15/1200px-telephone-keypad2svg.png)

**Example 1:**

**Input:** digits = "23"
**Output:** ["ad","ae","af","bd","be","bf","cd","ce","cf"]

**Example 2:**

**Input:** digits = "2"
**Output:** ["a","b","c"]

**Constraints:**

- `1 <= digits.length <= 4`
- `digits[i]` is a digit in the range `['2', '9']`.

```cpp
class Solution {
private:
    void helper(int idx,
                string &digits,
                string &curr,
                vector<string> &res,
                unordered_map<char, string> &mp) {

        if (idx == digits.length()) {
            res.push_back(curr);
            return;
        }

        string letters = mp[digits[idx]];

        for (int i = 0; i < letters.length(); i++) {
            curr.push_back(letters[i]);
            helper(idx + 1, digits, curr, res, mp);
            curr.pop_back();
        }
    }

public:
    vector<string> letterCombinations(string digits) {

        if (digits.empty()) {
            return {};
        }

        unordered_map<char, string> mp = {
            {'2', "abc"},
            {'3', "def"},
            {'4', "ghi"},
            {'5', "jkl"},
            {'6', "mno"},
            {'7', "pqrs"},
            {'8', "tuv"},
            {'9', "wxyz"}
        };

        vector<string> res;
        string curr;

        helper(0, digits, curr, res, mp);

        return res;
    }
};
```


For **LeetCode 17 (Letter Combinations of a Phone Number)**, the standard **backtracking/DFS** solution is generally considered the best because:

- Time complexity is optimal: (O(4^n \cdot n)) in the worst case (`7` and `9` have 4 letters).
    
- Space complexity is (O(n)) excluding the output.
    
- It's simple and easy to understand.
    

### Backtracking Solution

```cpp
class Solution {
private:
    void helper(int idx,
                string &digits,
                string &curr,
                vector<string> &res,
                vector<string> &mp) {

        if (idx == digits.size()) {
            res.push_back(curr);
            return;
        }

        string &letters = mp[digits[idx] - '0'];

        for (char ch : letters) {
            curr.push_back(ch);
            helper(idx + 1, digits, curr, res, mp);
            curr.pop_back();
        }
    }

public:
    vector<string> letterCombinations(string digits) {

        if (digits.empty()) return {};

        vector<string> mp = {
            "", "", "abc", "def", "ghi",
            "jkl", "mno", "pqrs", "tuv", "wxyz"
        };

        vector<string> res;
        string curr;

        helper(0, digits, curr, res, mp);

        return res;
    }
};
```

### Why use a `vector<string>` instead of `unordered_map`?

Instead of:

```cpp
unordered_map<char,string> mp;
```

use:

```cpp
vector<string> mp = {
    "", "", "abc", "def", ...
};
```

Then access letters in O(1):

```cpp
mp[digits[idx] - '0']
```

This is faster, uses less memory, and is the common interview solution.

### Iterative Alternative

You can also build combinations level by level:

```cpp
class Solution {
public:
    vector<string> letterCombinations(string digits) {

        if (digits.empty()) return {};

        vector<string> mp = {
            "", "", "abc", "def", "ghi",
            "jkl", "mno", "pqrs", "tuv", "wxyz"
        };

        vector<string> res = {""};

        for (char digit : digits) {
            vector<string> temp;

            for (string &s : res) {
                for (char ch : mp[digit - '0']) {
                    temp.push_back(s + ch);
                }
            }

            res = move(temp);
        }

        return res;
    }
};
```

Both are optimal. In interviews, the recursive backtracking version is usually preferred because it directly expresses the "choose a letter for each digit" decision process.
