```cpp
class Solution {
private:
    void helper(int idx,
                vector<int>& nums,
                vector<int>& curr,
                vector<vector<int>>& res) {

        if (idx == nums.size()) {
            res.push_back(curr);
            return;
        }

        // Take nums[idx]
        curr.push_back(nums[idx]);
        helper(idx + 1, nums, curr, res);

        // Don't take nums[idx]
        curr.pop_back();
        helper(idx + 1, nums, curr, res);
    }

public:
    vector<vector<int>> subsets(vector<int>& nums) {
        vector<vector<int>> res;
        vector<int> curr;

        helper(0, nums, curr, res);

        return res;
    }
};
```

**pass the entire original array and either keep the current element of remove it**

```cpp
class Solution {
private:
    void helper(int idx, vector<int>& nums, vector<vector<int>>& res) {
    if (idx == nums.size()) {
        res.push_back(nums);
        return;
    }

    // Keep current element
    helper(idx + 1, nums, res);

    // Remove current element
    int val = nums[idx];
    nums.erase(nums.begin() + idx);

    helper(idx, nums, res);

    nums.insert(nums.begin() + idx, val); // backtrack
}

public:
    vector<vector<int>> subsets(vector<int>& nums) {
        vector<vector<int>> res;
        vector<int> curr;

        helper(0, nums, res);

        return res;
    }
};
```
